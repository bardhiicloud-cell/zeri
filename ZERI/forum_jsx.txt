  return (
    <div className="h-[100dvh] w-full bg-slate-50 dark:bg-[var(--color-secondary)] font-sans overflow-hidden flex flex-col text-[var(--color-text)] dark:text-white dark:text-white relative transition-colors">
      <TopBar />
      
      {/* Top Floating Nav (TikTok Style) */}
      <div className="absolute top-0 inset-x-0 z-40 pt-20 md:pt-12 pb-4 flex items-center justify-center gap-6 bg-gradient-to-b from-white/90 dark:from-black/80 to-transparent dark:to-transparent pointer-events-none">

        <button 
          onClick={() => setActiveTab('feed')} 
          className={`pointer-events-auto font-bold text-[15px] transition-all drop-shadow-[0_4px_20px_-2px_rgba(0,0,0,0.05)] dark:shadow-[0_4px_20px_-2px_rgba(0,0,0,0.3)] ${activeTab === 'feed' ? 'text-[var(--color-text)] dark:text-white dark:text-white' : 'text-[var(--color-text-muted)] dark:text-[var(--color-text-muted)] hover:text-slate-700 dark:text-[var(--color-text-muted)] dark:text-white/60 dark:hover:text-[var(--color-text)] dark:text-white/80'}`}
        >{dt(`Debati`)}
          {activeTab === 'feed' && <div className="w-6 h-1 bg-slate-900 dark:bg-slate-800 rounded-full mx-auto mt-1" />}
        </button>
        
        <button 
          onClick={() => setActiveTab('profile')} 
          className={`pointer-events-auto font-bold text-[15px] transition-all drop-shadow-[0_4px_20px_-2px_rgba(0,0,0,0.05)] dark:shadow-[0_4px_20px_-2px_rgba(0,0,0,0.3)] ${activeTab === 'profile' ? 'text-[var(--color-text)] dark:text-white dark:text-white' : 'text-[var(--color-text-muted)] dark:text-[var(--color-text-muted)] hover:text-slate-700 dark:text-[var(--color-text-muted)] dark:text-white/60 dark:hover:text-[var(--color-text)] dark:text-white/80'}`}
        >{dt(`Profili`)}
          {activeTab === 'profile' && <div className="w-6 h-1 bg-slate-900 dark:bg-slate-800 rounded-full mx-auto mt-1" />}
        </button>
      </div>

      {/* Main Content Area */}
      <div className="flex-1 w-full relative h-full overflow-hidden">
        {activeTab === 'feed' && (
          <div className="w-full h-full overflow-y-auto bg-slate-50 dark:bg-[var(--color-secondary)] text-[var(--color-text)] dark:text-white dark:text-white hide-scrollbar pt-32 md:pt-24 pb-20">
            <div className="max-w-xl mx-auto w-full border-x border-slate-200/60 dark:border-slate-200/60 dark:border-zinc-800/80 min-h-full pb-32">
              
              {/* Create Text Post Input */}
              <div className="bg-transparent border-b border-slate-200/60 dark:border-slate-200/60 dark:border-zinc-800/80 p-4 mb-0">
                <div className="flex items-start gap-3">
                  <img src={currentUser?.avatar || 'https://ui-avatars.com/api/?name=U'} className="w-10 h-10 rounded-full object-cover shrink-0" alt="" />
                  <div className="flex-1">
                    <textarea
                      ref={textareaRef}
                      value={newPostContent}
                      onChange={(e) => setNewPostContent(e.target.value)}
                      placeholder={dt(`Çfarë po ndodh?`)}
                      className="w-full bg-transparent border-none text-[var(--color-text)] dark:text-white dark:text-white resize-none focus:ring-0 p-0 placeholder:text-[var(--color-text-muted)] dark:placeholder:text-[var(--color-text-muted)] dark:text-[var(--color-text-muted)] dark:text-zinc-500 text-lg sm:text-xl mt-1 mb-2 outline-none"
                      rows={newPostContent ? 3 : 1}
                    />
                    
                    <div className="flex items-center justify-between pt-3 border-t border-slate-200/60 dark:border-slate-200/60 dark:border-zinc-800/80">
                      <div className="flex items-center gap-2">
                        <button onClick={() => fileInputRef.current?.click()} className="p-2 text-rose-500 hover:bg-rose-500/10 rounded-full transition-colors">
                          <ImageIcon className="w-5 h-5" />
                        </button>
                        <input type="file" accept="image/*,video/*" className="hidden" ref={fileInputRef} onChange={handleFileSelect} />
                        
                        <button onClick={() => setShowPollCreator(!showPollCreator)} className="p-2 text-rose-500 hover:bg-rose-500/10 rounded-full transition-colors">
                          <SlidersHorizontal className="w-5 h-5" />
                        </button>
                        <button className="p-2 text-rose-500 hover:bg-rose-500/10 rounded-full transition-colors hidden sm:block">
                          <Smile className="w-5 h-5" />
                        </button>
                      </div>
                      <button 
                        onClick={handleCreatePost}
                        disabled={isSubmitting || (!newPostContent.trim() && !newPostImage && !pollQuestion)}
                        className="px-5 py-1.5 bg-rose-500 hover:bg-rose-600 disabled:opacity-50 text-[var(--color-text)] dark:text-white text-[15px] font-bold rounded-full transition-all"
                      >{dt(`Publiko`)}
                      </button>
                    </div>

                    {/* Poll Creator */}
                    {showPollCreator && (
                      <div className="mt-3 pt-3 border-t border-slate-200/60 dark:border-zinc-800 space-y-2">
                        <input type="text" placeholder={dt(`Bëj një pyetje...`)} className="w-full bg-white dark:bg-[var(--color-secondary)] border border-slate-200/60 dark:border-slate-200/60 dark:border-zinc-800 rounded-xl px-3 py-3 text-sm text-[var(--color-text)] dark:text-white dark:text-white focus:border-rose-500 focus:outline-none" value={pollQuestion || ''} onChange={e => setPollQuestion(e.target.value)} />
                        {(pollOptions || ['', '']).map((opt, i) => (
                          <input key={i} type="text" placeholder={`Opsioni ${i+1}`} className="w-full bg-white dark:bg-[var(--color-secondary)] border border-slate-200/60 dark:border-slate-200/60 dark:border-zinc-800 rounded-xl px-3 py-3 text-sm text-[var(--color-text)] dark:text-white dark:text-white focus:border-rose-500 focus:outline-none" value={opt} onChange={e => { const newOpts = [...pollOptions]; newOpts[i] = e.target.value; setPollOptions(newOpts); }} />
                        ))}
                      </div>
                    )}
                    {/* Selected File Preview */}
                    {newPostImage && (
                      <div className="mt-3 relative rounded-[1rem] overflow-hidden bg-slate-100 dark:bg-zinc-950 border border-slate-200/60 dark:border-slate-200/60 dark:border-zinc-800 p-1">
                         <span className="text-xs text-emerald-400 px-2">{dt(`Media e zgjedhur:`)} {newPostImage.name}</span>
                         <button onClick={() => setNewPostImage(null)} className="absolute top-1 right-1 p-1 bg-[var(--color-secondary)]/50 rounded-full"><X className="w-4 h-4 text-[var(--color-text)] dark:text-white" /></button>
                      </div>
                    )}
                  </div>
                </div>
              </div>

              {/* Feed of Text Posts */}
              {isLoading ? (
                Array.from({ length: 5 }).map((_, i) => (
                  <div key={i} className="border-b border-slate-200/60 dark:border-slate-200/60 dark:border-zinc-800/80 p-4 animate-pulse flex gap-3">
                    <div className="w-10 h-10 rounded-full bg-zinc-800 shrink-0"></div>
                    <div className="flex-1 space-y-3">
                      <div className="flex gap-2 items-center">
                        <div className="h-4 w-32 bg-zinc-800 rounded"></div>
                        <div className="h-3 w-16 bg-zinc-800/50 rounded"></div>
                      </div>
                      <div className="space-y-2">
                        <div className="h-3 w-full bg-zinc-800 rounded"></div>
                        <div className="h-3 w-5/6 bg-zinc-800 rounded"></div>
                        <div className="h-3 w-4/6 bg-zinc-800 rounded"></div>
                      </div>
                      <div className="h-32 w-full bg-zinc-800/50 rounded-[1rem] mt-3"></div>
                      <div className="flex gap-6 mt-4">
                        <div className="h-4 w-12 bg-zinc-800 rounded"></div>
                        <div className="h-4 w-12 bg-zinc-800 rounded"></div>
                        <div className="h-4 w-12 bg-zinc-800 rounded"></div>
                      </div>
                    </div>
                  </div>
                ))
              ) : posts.length === 0 ? (
                <div className="p-8 text-center text-[var(--color-text-muted)] dark:text-[var(--color-text-muted)] dark:text-zinc-500">{dt(`Nuk u gjetën postime.`)}</div>
              ) : posts.map(post => (
                <div key={post.id} className="bg-white dark:bg-[var(--color-secondary)] border-b border-slate-200/60 dark:border-slate-200 dark:border-zinc-900 p-5 transition-all hover:bg-slate-50 dark:bg-slate-800 dark:hover:bg-zinc-950 group"> 
                  <div className="flex gap-3">
                    <img 
                      src={post.authorAvatar} 
                      className="w-10 h-10 rounded-full object-cover shrink-0 cursor-pointer hover:opacity-80 transition-opacity" 
                      onClick={() => setProfileModalData({isOpen: true, authorId: post.authorId, authorName: post.authorName, authorAvatar: post.authorAvatar})}
                    />
                    <div className="flex-1 min-w-0">
                      {/* Header Row */}
                      <div className="flex items-center gap-1.5 text-[15px] mb-0.5">
                        <span 
                          className="font-bold text-[var(--color-text)] dark:text-white dark:text-white hover:underline cursor-pointer truncate max-w-[150px] sm:max-w-[200px] text-[15px]"
                          onClick={() => setProfileModalData({isOpen: true, authorId: post.authorId, authorName: post.authorName, authorAvatar: post.authorAvatar})}
                        >
                          {post.authorName}
                        </span>
                        {/* Twitter Handle Mock */}
                        <span className="text-[var(--color-text-muted)] dark:text-[var(--color-text-muted)] dark:text-zinc-500 truncate">@{(post.authorName || 'user').replace(/\s+/g, '').toLowerCase()}</span>
                        <span className="text-[var(--color-text-muted)] dark:text-[var(--color-text-muted)] dark:text-zinc-500">·</span>
                        <span className="text-[var(--color-text-muted)] dark:text-[var(--color-text-muted)] dark:text-zinc-500 text-sm whitespace-nowrap hover:underline cursor-pointer">
                          {new Date(post.createdAt).toLocaleDateString('sq-AL', { month: 'short', day: 'numeric' })}
                        </span>
                      </div>
                      
                      {/* Content */}
                      <div className="text-[15px] text-slate-700 dark:text-[var(--color-text-muted)] dark:text-white/90 whitespace-pre-wrap leading-relaxed mb-3 break-words mt-1">
                        <PostFormatter content={post.content} />
                      </div>
                      
                      {/* Media */}
                      {post.imageUrl && (
                        <div className="rounded-[1.25rem] overflow-hidden border border-slate-200/60 dark:border-zinc-800 mb-3 bg-[var(--color-secondary)]/50">
                          <img src={post.imageUrl} className="w-full max-h-[500px] object-cover" alt="" />
                        </div>
                      )}
                      
                      {/* Poll */}
                      {post.poll && (
                        <div className="border border-slate-200/60 dark:border-zinc-800 rounded-[1.25rem] p-4 mb-4 bg-slate-50 dark:bg-zinc-900/30 shadow-[inset_0_1px_1px_rgba(255,255,255,0.05)]">
                          <span className="font-bold text-[15px] mb-3 block text-[var(--color-text)] dark:text-white">{post.poll.question}</span>
                          <div className="space-y-2">
                            {(post.poll.options || []).map((opt, i) => (
                              <button key={i} className="w-full bg-white dark:bg-zinc-900/50 hover:bg-blue-50 dark:hover:bg-blue-500/10 border border-slate-200/60 dark:border-zinc-700 hover:border-blue-500 text-slate-700 dark:text-[var(--color-text-muted)] dark:text-white hover:text-blue-600 dark:hover:text-blue-400 font-bold rounded-[1rem] py-3 px-4 text-[13px] transition-all text-left flex justify-between shadow-sm dark:shadow-none">
                                <span>{opt.text}</span>
                                <span>{Math.floor(Math.random() * 20)}%</span> {/* Mock % */}
                              </button>
                            ))}
                          </div>
                          <span className="text-xs text-[var(--color-text-muted)] dark:text-[var(--color-text-muted)] dark:text-zinc-500 block mt-3">{Math.floor(Math.random() * 100) + 10}{dt(`vota · 2 ditë të mbetura`)}</span>
                        </div>
                      )}
                      
                      {/* Action Row */}
                      <div className="flex items-center justify-between text-[var(--color-text-muted)] dark:text-[var(--color-text-muted)] dark:text-zinc-500 max-w-md pr-4 mt-3 border-t border-slate-200 dark:border-zinc-900 pt-3">
                        <button 
                          onClick={() => setActiveCommentPost(activeCommentPost === post.id ? null : post.id)} 
                          className={`flex items-center gap-2 text-[13px] group transition-colors ${activeCommentPost === post.id ? 'text-blue-400' : 'hover:text-blue-400'}`}
                        >
                          <div className={`p-2 rounded-full transition-colors ${activeCommentPost === post.id ? 'bg-blue-400/10' : 'group-hover:bg-blue-400/10'}`}>
                            <MessageCircle className={`w-4 h-4 ${activeCommentPost === post.id ? 'fill-blue-400/20' : ''}`} />
                          </div>
                          <span>{post.comments?.length || 0}</span>
                        </button>
                        
                        <button 
                          onClick={() => handleRepost(post.id)}
                          className={`flex items-center gap-2 text-[13px] group transition-colors ${(post.reposts || []).includes(currentUser?.id) ? 'text-emerald-400' : 'hover:text-emerald-400'}`}
                        >
                          <div className={`p-2 rounded-full transition-colors ${(post.reposts || []).includes(currentUser?.id) ? 'bg-emerald-400/10' : 'group-hover:bg-emerald-400/10'}`}>
                            <RefreshCw className={`w-4 h-4 ${(post.reposts || []).includes(currentUser?.id) ? 'text-emerald-400' : ''}`} />
                          </div>
                          <span>{(post.reposts || []).length}</span>
                        </button>
                        
                        <button 
                          onClick={() => handleReaction(post.id, 'like')} 
                          className={`flex items-center gap-2 text-[13px] group transition-colors ${(post.likes || []).includes(currentUser?.id) ? 'text-rose-500' : 'hover:text-rose-500'}`}
                        >
                          <div className={`p-2 rounded-full transition-colors ${(post.likes || []).includes(currentUser?.id) ? 'bg-rose-500/10' : 'group-hover:bg-rose-500/10'}`}>
                            <Heart className={`w-4 h-4 ${(post.likes || []).includes(currentUser?.id) ? 'fill-rose-500 text-rose-500' : ''}`} />
                          </div>
                          <span className={(post.likes || []).includes(currentUser?.id) ? 'text-rose-500' : 'group-hover:text-rose-500'}>
                            {(post.likes || []).length}
                          </span>
                        </button>
                        
                        <button 
                          onClick={() => {
                            if (navigator.share) {
                              navigator.share({
                                title: `Postimi nga ${post.authorName}`,
                                text: (post.content || '').substring(0, 100) + '...',
                                url: window.location.href,
                              }).catch(() => {});
                            } else {
                              navigator.clipboard.writeText(window.location.href);
                              toast.success('Linku u kopjua në clipboard!');
                            }
                          }}
                          className="flex items-center gap-2 text-[13px] hover:text-blue-400 group transition-colors"
                        >
                          <div className="p-2 rounded-full group-hover:bg-blue-400/10 transition-colors">
                            <Share2 className="w-4 h-4" />
                          </div>
                        </button>
                      </div>
                      
                      {/* Comments Section Overlay */}
                      <AnimatePresence>
                        {activeCommentPost === post.id && (
                          <motion.div 
                            initial={{ opacity: 0, height: 0 }} 
                            animate={{ opacity: 1, height: 'auto' }} 
                            exit={{ opacity: 0, height: 0 }}
                            className="overflow-hidden"
                          >
                            <div className="mt-2 pt-3 border-t border-slate-200/60 dark:border-slate-200/60 dark:border-zinc-800/80">
                              <div className="space-y-4 mb-4">
                                {(post.comments || []).length === 0 && (
                                  <p className="text-[var(--color-text-muted)] dark:text-[var(--color-text-muted)] dark:text-zinc-500 text-sm italic text-center py-4">{dt(`Nuk ka ende komente. Bëhu i pari!`)}</p>
                                )}
                                {(post.comments || []).map((comment: any) => (
                                  <div key={comment.id} className="flex gap-3">
                                    <img src={comment.authorAvatar || 'https://ui-avatars.com/api/?name=U'} className="w-8 h-8 rounded-full object-cover shrink-0" alt="" />
                                    <div className="flex-1">
                                      <div className="bg-zinc-900/50 rounded-[1.25rem] px-3 py-2 border border-slate-200/60 dark:border-zinc-800">
                                        <div className="flex items-center gap-2 mb-1">
                                          <span className="font-bold text-xs text-[var(--color-text)] dark:text-white">{(comment.authorName || 'user')}</span>
                                          <span className="text-[10px] text-[var(--color-text-muted)] dark:text-[var(--color-text-muted)] dark:text-zinc-500">{new Date(comment.createdAt).toLocaleDateString('sq-AL')}</span>
                                        </div>
                                        <p className="text-[13px] text-slate-700 dark:text-[var(--color-text-muted)] dark:text-white/90 whitespace-pre-wrap">{comment.content}</p>
                                      </div>
                                      <div className="flex items-center gap-4 mt-1 px-2 text-[11px] font-bold text-[var(--color-text-muted)] dark:text-[var(--color-text-muted)] dark:text-zinc-500">
                                        <button 
                                          className="hover:text-blue-400 transition-colors"
                                          onClick={() => setReplyingCommentId(replyingCommentId === comment.id ? null : comment.id)}
                                        >{dt(`Përgjigju`)}
                                        </button>
                                        <button className="hover:text-rose-400 transition-colors">{dt(`Pëlqe`)}</button>
                                      </div>
                                      
                                      {/* Replies List */}
                                      {comment.replies && comment.replies.length > 0 && (
                                        <div className="mt-3 space-y-3 pl-2 border-l-2 border-slate-200/60 dark:border-zinc-800/50">
                                          {(comment.replies || []).map((reply: any) => (
                                            <div key={reply.id} className="flex gap-2">
                                              <img src={reply.authorAvatar || 'https://ui-avatars.com/api/?name=U'} className="w-6 h-6 rounded-full object-cover shrink-0" alt="" />
                                              <div className="flex-1 bg-slate-50 dark:bg-zinc-900/30 rounded-[1.25rem] px-3 py-2 border border-slate-200/60 dark:border-zinc-800/50">
                                                <div className="flex items-center gap-2 mb-1">
                                                  <span className="font-bold text-[11px] text-slate-700 dark:text-[var(--color-text-muted)] dark:text-white/90">{(reply.authorName || 'user')}</span>
                                                  <span className="text-[9px] text-[var(--color-text-muted)] dark:text-[var(--color-text-muted)] dark:text-zinc-500">{new Date(reply.createdAt).toLocaleDateString('sq-AL')}</span>
                                                </div>
                                                <p className="text-xs text-[var(--color-text)] dark:text-white/80 whitespace-pre-wrap">{reply.content}</p>
                                              </div>
                                            </div>
                                          ))}
                                        </div>
                                      )}

                                      {/* Reply Input Form */}
                                      {replyingCommentId === comment.id && (
                                        <div className="mt-3 flex gap-2 pl-2 border-l-2 border-blue-500/50">
                                          <img src={currentUser?.avatar || 'https://ui-avatars.com/api/?name=U'} className="w-6 h-6 rounded-full object-cover shrink-0" alt="" />
                                          <div className="flex-1 flex bg-zinc-900 border border-zinc-700 rounded-full overflow-hidden focus-within:border-blue-500 transition-colors">
                                            <input 
                                              type="text" 
                                              value={replyContent}
                                              onChange={(e) => setReplyContent(e.target.value)}
                                              placeholder={`Përgjigju ${(comment.authorName || 'user')}...`}
                                              className="flex-1 bg-transparent text-xs text-[var(--color-text)] dark:text-white px-3 py-1.5 focus:outline-none"
                                              onKeyDown={(e) => {
                                                if (e.key === 'Enter' && !e.shiftKey) {
                                                  e.preventDefault();
                                                  handleAddReply(post.id, comment.id);
                                                }
                                              }}
                                            />
                                            <button 
                                              onClick={() => handleAddReply(post.id, comment.id)}
                                              disabled={!replyContent.trim()}
                                              className="px-3 text-blue-500 hover:text-blue-400 disabled:opacity-50 transition-colors"
                                            >
                                              <Send className="w-3.5 h-3.5" />
                                            </button>
                                          </div>
                                        </div>
                                      )}
                                    </div>
                                  </div>
                                ))}
                              </div>
                              <div className="flex items-start gap-3 mt-4">
                                <img src={currentUser?.avatar || 'https://ui-avatars.com/api/?name=U'} className="w-8 h-8 rounded-full object-cover shrink-0 mt-1" alt="" />
                                <div className="flex-1 flex flex-col relative">
                                  <textarea 
                                    value={newCommentContent}
                                    onChange={e => setNewCommentContent(e.target.value)}
                                    placeholder={dt(`Posto përgjigjen tënde...`)} 
                                    className="w-full bg-[var(--color-secondary)] border border-slate-200/60 dark:border-zinc-800 rounded-[1rem] px-4 py-2.5 text-sm text-[var(--color-text)] dark:text-white focus:border-blue-500 focus:outline-none resize-none min-h-[44px]"
                                    rows={newCommentContent ? 2 : 1}
                                  />
                                  <div className="flex justify-between items-center mt-2">
                                    <div className="flex gap-2">
                                      <button className="p-1.5 text-blue-400 hover:bg-blue-400/10 rounded-full"><ImageIcon className="w-4 h-4"/></button>
                                      <button className="p-1.5 text-blue-400 hover:bg-blue-400/10 rounded-full"><Smile className="w-4 h-4"/></button>
                                    </div>
                                    <button 
                                      onClick={() => handleAddComment(post.id)}
                                      disabled={!newCommentContent.trim() || isSubmitting}
                                      className="px-4 py-1.5 bg-blue-500 hover:bg-blue-600 disabled:opacity-50 text-[var(--color-text)] dark:text-white font-bold text-xs rounded-full"
                                    >{dt(`Përgjigju`)}
                                    </button>
                                  </div>
                                </div>
                              </div>
                            </div>
                          </motion.div>
                        )}
                      </AnimatePresence>
                      
                    </div>
                  </div>
                </div>
              ))}
            </div>
          </div>
        )}

        

        {activeTab === 'profile' && (
          <div className="w-full h-full overflow-y-auto bg-[var(--color-secondary)] text-[var(--color-text)] dark:text-white pt-16 pb-20 hide-scrollbar">
            <ForumProfileTab currentUser={currentUser} setActiveTab={setActiveTab} />
          </div>
        )}
      </div>

      

      <ForumProfileModal 
        isOpen={profileModalData.isOpen} 
        onClose={() => setProfileModalData({ ...profileModalData, isOpen: false })} 
        authorId={profileModalData.authorId} 
        authorName={profileModalData.authorName} 
        authorAvatar={profileModalData.authorAvatar} 
        posts={posts.filter(p => p.authorId === profileModalData.authorId)}
      />
    </div>
  );
}
